cmake_minimum_required(VERSION 3.22.1)

# 需要 C：libogg / libvorbis 是 C 库
project(rlvm_android CXX C)

# task.md：C++17 是硬性要求（libreallive / machine 均依赖）。
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ===========================================================================
# 路径解析
# ===========================================================================

# RLVM 上游源码保留在原处，不复制（见 docs/DECISIONS.md D-006）。
# CMAKE_CURRENT_SOURCE_DIR = <repo>/app/src/main/cpp，向上四级即仓库根。
# 仓库根本身名为 rlvm-release-0.14，上游源码在其下的 rlvm-release-0.14/ 子目录。
get_filename_component(RLVM_SOURCE_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../../rlvm-release-0.14" ABSOLUTE)
if(NOT EXISTS "${RLVM_SOURCE_DIR}/src/machine/rlmachine.cc")
    message(FATAL_ERROR "找不到 RLVM 上游源码: ${RLVM_SOURCE_DIR}")
endif()

# Boost 源码树（见 docs/DECISIONS.md D-010）。
# 优先级：-DBOOST_SOURCE_DIR=... > 环境变量 BOOST_SOURCE_DIR > 本机默认路径。
if(NOT BOOST_SOURCE_DIR AND DEFINED ENV{BOOST_SOURCE_DIR})
    file(TO_CMAKE_PATH "$ENV{BOOST_SOURCE_DIR}" BOOST_SOURCE_DIR)
endif()
if(NOT BOOST_SOURCE_DIR AND EXISTS "E:/boost-1.92.0/boost/version.hpp")
    set(BOOST_SOURCE_DIR "E:/boost-1.92.0")
endif()
if(NOT EXISTS "${BOOST_SOURCE_DIR}/boost/version.hpp")
    message(FATAL_ERROR "找不到 Boost 源码树，请用 -DBOOST_SOURCE_DIR=<boost 根目录> 指定")
endif()

message(STATUS "RLVM source : ${RLVM_SOURCE_DIR}")
message(STATUS "Boost source: ${BOOST_SOURCE_DIR}")

# 第三方源码目录（由 tools/setup_third_party.ps1 获取，已在 .gitignore 中排除）
if(NOT THIRD_PARTY_DIR AND DEFINED ENV{RLVM_THIRD_PARTY_DIR})
    file(TO_CMAKE_PATH "$ENV{RLVM_THIRD_PARTY_DIR}" THIRD_PARTY_DIR)
endif()
if(NOT THIRD_PARTY_DIR)
    get_filename_component(THIRD_PARTY_DIR
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party" ABSOLUTE)
endif()

set(OGG_SOURCE_DIR "${THIRD_PARTY_DIR}/ogg-1.3.5")
set(VORBIS_SOURCE_DIR "${THIRD_PARTY_DIR}/vorbis-1.3.7")
if(NOT EXISTS "${OGG_SOURCE_DIR}/include/ogg/ogg.h")
    message(FATAL_ERROR "找不到 libogg 源码: ${OGG_SOURCE_DIR}（先运行 tools/setup_third_party.ps1）")
endif()
if(NOT EXISTS "${VORBIS_SOURCE_DIR}/include/vorbis/vorbisfile.h")
    message(FATAL_ERROR "找不到 libvorbis 源码: ${VORBIS_SOURCE_DIR}")
endif()
message(STATUS "libogg      : ${OGG_SOURCE_DIR}")
message(STATUS "libvorbis   : ${VORBIS_SOURCE_DIR}")

# ===========================================================================
# libogg：include/ogg/config_types.h 由 configure/cmake 生成，这里直接生成一份。
# Android 必然具备 inttypes.h / stdint.h / sys/types.h 与定长整型。
# ===========================================================================
set(OGG_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${OGG_GENERATED_DIR}/ogg")
file(WRITE "${OGG_GENERATED_DIR}/ogg/config_types.h"
"#ifndef __CONFIG_TYPES_H__
#define __CONFIG_TYPES_H__
/* Generated by app/src/main/cpp/CMakeLists.txt for the Android build. */
#define INCLUDE_INTTYPES_H 1
#define INCLUDE_STDINT_H 1
#define INCLUDE_SYS_TYPES_H 1
#include <inttypes.h>
#include <stdint.h>
#include <sys/types.h>
typedef int16_t  ogg_int16_t;
typedef uint16_t ogg_uint16_t;
typedef int32_t  ogg_int32_t;
typedef uint32_t ogg_uint32_t;
typedef int64_t  ogg_int64_t;
typedef uint64_t ogg_uint64_t;
#endif
")

add_library(ogg STATIC
    "${OGG_SOURCE_DIR}/src/bitwise.c"
    "${OGG_SOURCE_DIR}/src/framing.c"
)
target_include_directories(ogg PUBLIC
    "${OGG_SOURCE_DIR}/include"
    "${OGG_GENERATED_DIR}"
)

# ===========================================================================
# libvorbis：源码不需要 configure 生成的 config.h，可直接编译。
# RLVM 通过 <vorbis/vorbisfile.h> 的 ov_* 接口解码 OVK/KOE 语音。
# 只编解码器：编码器部分（vorbisenc / psy / psytune / analysis / barkmel / tone）
# 依赖 lib/modes/*.h 这类构建时生成的头文件，而 RLVM 完全用不到编码功能。
# ===========================================================================
set(_vorbis_lib "${VORBIS_SOURCE_DIR}/lib")
add_library(vorbis STATIC
    "${_vorbis_lib}/bitrate.c"
    "${_vorbis_lib}/block.c"
    "${_vorbis_lib}/codebook.c"
    "${_vorbis_lib}/envelope.c"
    "${_vorbis_lib}/floor0.c"
    "${_vorbis_lib}/floor1.c"
    "${_vorbis_lib}/info.c"
    "${_vorbis_lib}/lookup.c"
    "${_vorbis_lib}/lpc.c"
    "${_vorbis_lib}/lsp.c"
    "${_vorbis_lib}/mapping0.c"
    "${_vorbis_lib}/mdct.c"
    "${_vorbis_lib}/misc.c"
    # psy.c 属于编码器心理学模型，但它内部并未使用 modes/*.h，
    # 且 res0.c 等仍会引用 _vp_* 符号，故保留。
    "${_vorbis_lib}/psy.c"
    "${_vorbis_lib}/registry.c"
    "${_vorbis_lib}/res0.c"
    "${_vorbis_lib}/sharedbook.c"
    "${_vorbis_lib}/smallft.c"
    "${_vorbis_lib}/synthesis.c"
    "${_vorbis_lib}/vorbisfile.c"
    "${_vorbis_lib}/window.c"
)
target_include_directories(vorbis PUBLIC "${VORBIS_SOURCE_DIR}/include")
target_link_libraries(vorbis PUBLIC ogg)

    list(APPEND RLVM_CORE_SOURCES_ABS "${RLVM_SOURCE_DIR}/${_rlvm_src}")

# ===========================================================================
# 目标
# ===========================================================================
add_library(rlvm SHARED
    native-bridge.cpp
    # Android 平台后端（替代上游的 SDLSystem）
    android/android_graphics.cpp
    android/android_system.cpp
    android/jni_saf_backend.cpp
    android/saf_file_system.cpp
    # SDL 音频转换的最小兼容层：供 vendor/xclannad/wavfile.cc 使用
    compat/sdl_shim/sdl_audio_compat.cpp
    ${BOOST_COMPILED_SOURCES}
    ${RLVM_CORE_SOURCES_ABS}
)

target_include_directories(rlvm PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"                 # "android/xxx.h" 这类本项目内部引用
    "${RLVM_SOURCE_DIR}/src"      # "libreallive/xxx.h" 这类引用
    "${RLVM_SOURCE_DIR}/vendor"   # "utf8cpp/xxx.h"、"lru_cache.hpp" 等
    "${BOOST_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/compat/sdl_shim"   # 提供 <SDL/SDL_mixer.h>
)

target_compile_definitions(rlvm PRIVATE
    CASE_SENSITIVE_FILESYSTEM   # Android 文件系统区分大小写（与上游 SConstruct 一致）
    _THREAD_SAFE
    # Boost 1.92 的 Boost.Filesystem 私有源文件（libs/filesystem/src/atomic_tools.hpp）
    # 使用 std::atomic_ref，而 NDK r28c 的 libc++ 并未实现该 C++20 特性。
    # Boost 为此提供了官方开关：定义后内部改用非原子访问。
    # 该宏只出现在 Boost.Filesystem 的私有源文件里，公共头文件不受影响，故不构成 ODR 风险。
    BOOST_FILESYSTEM_SINGLE_THREADED
)

target_compile_options(rlvm PRIVATE
    -fexceptions
    -frtti
    -Wall
    -Wno-sign-compare          # 与上游 SConscript 一致：抑制老代码与第三方噪声
    -Wno-narrowing
)

    vorbis                      # 语音解码（OVK/KOE Ogg Vorbis）；ogg 由其 PUBLIC 传递
    ${log_lib}
    ${android_lib}
    ${z_lib}
)
